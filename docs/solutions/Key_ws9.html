# create bray curtis dissimilarity matrix

# define trajectory
d = defineTrajectories(b_dist, sites = site, surveys =survey)

  1. Plot the cumulative length at each time step for each site.
tll = tll |> 
      group_by(site) |> 
      mutate(dist_cuml = cumsum(len))
